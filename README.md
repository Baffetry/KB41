-- Должность (id [PK], название)
CREATE TABLE IF NOT EXISTS job_title
(
	id SMALLINT GENERATED ALWAYS AS IDENTITY,
	name VARCHAR(30) NOT NULL,

	CONSTRAINT PK_job_title_id PRIMARY KEY(id)
);

-- Факультет (id [PK], название, телефон, адрес)
CREATE TABLE IF NOT EXISTS faculty 
(
	id SMALLINT GENERATED ALWAYS AS IDENTITY,
	name VARCHAR(255) NOT NULL,
	phone VARCHAR(20) NOT NULL,
	address TEXT NOT NULL,

	CONSTRAINT PK_faculty_id PRIMARY KEY(id)
);

-- Кафедра (id [PK], название, телефон, адрес, факультет_id [FK])
CREATE TABLE IF NOT EXISTS department
(
	id SMALLINT GENERATED ALWAYS AS IDENTITY,
	name VARCHAR(255) NOT NULL,
	phone VARCHAR(20) NOT NULL,
	address TEXT NOT NULL,
	faculty_id SMALLINT NOT NULL,

	CONSTRAINT PK_department_id PRIMARY KEY(id),
	CONSTRAINT FK_department_faculty_id FOREIGN KEY(faculty_id)
		REFERENCES faculty(id) ON DELETE RESTRICT
);

-- Помещение (номер [PK], телефон, кафедра_id [FK], факультет_id [FK])
CREATE TABLE IF NOT EXISTS room
(
	number INT NOT NULL,
	phone VARCHAR(20),
	faculty_id SMALLINT NOT NULL,
	department_id SMALLINT,

	CONSTRAINT PK_room_number PRIMARY KEY(number),
	CONSTRAINT FK_room_faculty_id FOREIGN KEY(faculty_id)
		REFERENCES faculty(id) ON DELETE RESTRICT,
	CONSTRAINT FK_room_department_id FOREIGN KEY(department_id)
		REFERENCES department(id) ON DELETE SET NULL	
);

-- Сотрудник (id [PK], фамилия, имя, отчество, телефон, помещение_id [FK])
CREATE TABLE IF NOT EXISTS employee
(
	id INT GENERATED ALWAYS AS IDENTITY,
	first_name VARCHAR(100) NOT NULL,
	last_name VARCHAR(100) NOT NULL,
	middle_name VARCHAR(100),
	phone VARCHAR(20),
	room_id INT,

	CONSTRAINT PK_employee_id PRIMARY KEY(id),
	CONSTRAINT FK_employee_room_id FOREIGN KEY(room_id)
		REFERENCES room(number) ON DELETE SET NULL ON UPDATE CASCADE
);

-- Сотрудник_факультет (сотрудник_id [FK], факультет_id [FK], должность_id [FK])
CREATE TABLE IF NOT EXISTS employee_faculty
(
	employee_id INT NOT NULL,
	faculty_id SMALLINT NOT NULL,
	job_title_id SMALLINT NOT NULL,

	CONSTRAINT PK_employee_faculty PRIMARY KEY(employee_id, faculty_id, job_title_id),
	CONSTRAINT FK_employee_faculty_employee_id FOREIGN KEY(employee_id)
		REFERENCES employee(id) ON DELETE CASCADE,
	CONSTRAINT FK_employee_faculty_faculty_id FOREIGN KEY(faculty_id)
		REFERENCES faculty(id) ON DELETE CASCADE,
	CONSTRAINT FK_employee_faculty_job_title_id FOREIGN KEY(job_title_id)
		REFERENCES job_title(id) ON DELETE RESTRICT
);

-- Сотрудник_кафедра (сотрудник_id [FK], кафедра_id [FK], должность_id [FK])
CREATE TABLE IF NOT EXISTS employee_department
(
	employee_id INT NOT NULL,
	department_id SMALLINT NOT NULL,
	job_title_id SMALLINT NOT NULL,

	CONSTRAINT PK_employee_department PRIMARY KEY(employee_id, department_id, job_title_id),
	CONSTRAINT FK_employee_department_employee_id FOREIGN KEY(employee_id)
		REFERENCES employee(id) ON DELETE CASCADE,
	CONSTRAINT FK_employee_department_department_id FOREIGN KEY(department_id)
		REFERENCES department(id) ON DELETE CASCADE,
	CONSTRAINT FK_employee_department_job_title_id FOREIGN KEY(job_title_id)
		REFERENCES job_title(id) ON DELETE RESTRICT
);
